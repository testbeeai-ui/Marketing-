Custom Event Tracking

Facebook/Instagram Custom Events


<script>
// Track Purchase
fbq('track', 'Purchase', {
    value: 29.99,
    currency: 'USD',
    content_name: 'Product Name'
});

// Track Add to Cart
fbq('track', 'AddToCart', {
    value: 19.99,
    currency: 'USD'
});

// Track Custom Event
fbq('trackCustom', 'CustomEventName', {
    customParameter: 'value'
});
</script>
